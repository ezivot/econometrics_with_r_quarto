1  Conditional Expectation and Projection

In this chapter I replicate the examples from Hansen’s Chapter 2: Conditional Expectation and Projection.

The R packages used in the chapter are HansenData. Make sure these packages are installed an loaded prior to running the code in this chapter.

1.3 Distribution of Hourly Wages

Code 
# kernel density estimate of pdf
wage.density = density(wage, bw=5)
par(mfrow=c(1,2))
# empirical cdf
plot.ecdf(wage, main="CDF of wage", xlab="Dollars per Hour", 
            lwd=2, col="blue")
# estimate of pdf
plot(wage.density, main="pdf of wage", xlim=c(0, 100), xlab="Dollars per Hour",
       lwd=2, col="blue")

1.8 Conditional Distributions for Men and Women by race

Code 
# race = 1 = white, race = 2 = black
lwageWhiteWomen.density = density(lwage[female==1 & race==1], bw=0.1)
lwageWhiteMen.density = density(lwage[female==0 & race==1], bw=0.1)
lwageBlackWomen.density = density(lwage[female==1 & race==2], bw=0.1)
lwageBlackMen.density = density(lwage[female==0 & race==2], bw=0.1)
plot(lwageWhiteWomen.density, main="pdf of log(wage) for Men and Women by race", 
     xlab="Log Dollars per Hour", lwd=2, xlim = c(1, 6), ylim=c(0, 0.8))
points(lwageWhiteMen.density, type="l", col="blue", lwd=2)
points(lwageBlackWomen.density, type="l", col="green", lwd=2)
points(lwageBlackMen.density, type="l", col="red", lwd=2)
legend("topright", legend=c("White Women", "White Men", "Black Women", "Black Men"), 
       lty="solid", lwd=2, col=c("black", "blue", "green", "red"))

1.10 Conditional Mean Function

Compute conditional means of log wages for men and women conditional on education level

Code 
elevels = sort(unique(education))
cmean.mat = matrix(0, length(elevels), 2)
colnames(cmean.mat) = c("White.Male", "White.Female")
rownames(cmean.mat) = as.character(elevels)

for (i in 1:length(elevels)) {
  cmean.mat[i, 1] = mean(lwage[female==0 & race==1 & education==elevels[i]])
  cmean.mat[i, 2] = mean(lwage[female==1 & race==1 & education==elevels[i]])
}

1.11 Plot Conditional Mean Function

Code 
plot(elevels, cmean.mat[, 1], type="b", pch=16, lwd=2, 
     ylim=c(1.75,4), ylab="Log Dollars per Hour", xlab="Years of Education")
points(elevels, cmean.mat[, 2], type="b", pch=16, lwd=2, col="blue")
legend("topleft", legend=c("Men", "Women"), lty="solid", pch=16, lwd=2, col=c("black", "blue"))
